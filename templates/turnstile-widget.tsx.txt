  <>

  </>

/* (3) LOGIN CLIENT-SIDE (sem <form> action):
   No submit, pegue o token e valide antes do signInWithPassword: */
const token = (window as unknown as { turnstile?: { getResponse: () => string } }).turnstile?.getResponse?.();
if (!token) { setError("Confirme que você não é um robô."); return; }
// ... enviar { token } para POST /api/auth/turnstile e checar res.ok; depois:
try { (window as unknown as { turnstile?: { reset: () => void } }).turnstile?.reset?.(); } catch {}
